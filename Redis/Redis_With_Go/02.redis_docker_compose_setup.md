services:
  redis:
    image: redis:7-alpine
    container_name: hr360-redis
    restart: unless-stopped
    ports:
      - "6379:6379"
    command: >
      redis-server
      --requirepass ${REDIS_PASSWORD:-secret123}
      --maxmemory 256mb
      --maxmemory-policy allkeys-lru
      --loglevel notice
    volumes:
      - redis_data:/data
    networks:
      - redis-net
    healthcheck:
      test: ["CMD", "redis-cli", "-a", "${REDIS_PASSWORD:-secret123}", "ping"]
      interval: 10s
      timeout: 5s
      retries: 5

  redis-ui:
    image: redislabs/redisinsight:latest
    container_name: hr360-redis-ui
    restart: unless-stopped
    ports:
      - "5540:5540"
    volumes:
      - redisinsight_data:/data
    networks:
      - redis-net
    depends_on:
      redis:
        condition: service_healthy
